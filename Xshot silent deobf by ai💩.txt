-- Calculation Helper for Shot Direction (with automated spread simulation)
local function GetShotDirection(Origin, TargetPos, SpreadFactor)
    local Dir = (TargetPos - Origin).Unit
    if SpreadFactor and SpreadFactor > 0 then
        local SpreadModifier = SpreadFactor * ((TargetPos - Origin).Magnitude / 100)
        local SpreadX = (math.random() * 2 - 1) * SpreadModifier
        local SpreadY = (math.random() * 2 - 1) * SpreadModifier
        local SpreadZ = (math.random() * 2 - 1) * SpreadModifier
        
        Dir = ((TargetPos + Vector3.new(SpreadX, SpreadY, SpreadZ)) - Origin).Unit
    end
    return Dir
end
